cmake_minimum_required(VERSION 3.16)
project(cb_thumbnail_component LANGUAGES CXX)
if(NOT WIN32 OR NOT FFMPEG_DIR)
  message(FATAL_ERROR "Windows and FFMPEG_DIR are required")
endif()
set(runner "${CMAKE_CURRENT_LIST_DIR}/../../../cb_file_manager/windows/runner")
add_executable(thumbnail_component thumbnail_component.cpp "${runner}/ffmpeg_thumbnail_helper.cpp")
target_include_directories(thumbnail_component PRIVATE "${runner}" "${FFMPEG_DIR}/include")
target_link_directories(thumbnail_component PRIVATE "${FFMPEG_DIR}/lib")
target_link_libraries(thumbnail_component PRIVATE avformat avcodec avutil swscale gdiplus shlwapi)
target_compile_features(thumbnail_component PRIVATE cxx_std_17)
target_compile_definitions(thumbnail_component PRIVATE NOMINMAX _CRT_SECURE_NO_WARNINGS)
